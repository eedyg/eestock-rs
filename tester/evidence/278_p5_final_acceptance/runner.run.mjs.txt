const MIME = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.png': 'image/png', '.svg': 'image/svg+xml' };
const server = http.createServer((req, res) => {
  const u = new URL(req.url, 'http://x');
  let p = path.join(dist, decodeURIComponent(u.pathname));
  if (u.pathname === '/') p = path.join(dist, 'index.html');
  fs.readFile(p, (e, b) => {
    if (e) { res.statusCode = 404; res.end('nf'); return; }
    res.setHeader('content-type', MIME[path.extname(p)] ?? 'application/octet-stream');
    res.end(b);
  });
});
await new Promise((r) => server.listen(0, '127.0.0.1', r));
const PORT = server.address().port;
const base = `http://127.0.0.1:${PORT}/`;

const browser = await chromium.launch();
const results = [];
const errors = [];
const netAll = [];

async function openPage(params) {
  const avail = Number(new URLSearchParams(params).get('avail') ?? 600);
  const page = await browser.newPage({ viewport: { width: 1440, height: Math.max(900, avail + 320) }, deviceScaleFactor: 1 });
  page.on('pageerror', (e) => errors.push(`[${params}] ${e.message}`));
  page.on('request', (r) => { const u = r.url(); if (!u.startsWith(base)) netAll.push(u); });
  await page.goto(base + '?' + params, { waitUntil: 'load' });
  await page.waitForFunction(() => !!(window.__p5 && document.querySelector('[data-mp-stack]')), null, { timeout: 60_000 });
  await page.waitForFunction(() => Object.keys(window.__p5.domHeights()).length > 0 && window.__p5.domHeights()[Object.keys(window.__p5.domHeights())[0]] > 0, null, { timeout: 60_000 });
  await page.waitForTimeout(1200);
  return page;
}
const measure = (page) => page.evaluate(() => window.__p5.measure());
async function drag(page, key, dy) {
  const box = await page.locator(`[data-mp-separator="${key}"]`).boundingBox();
  const x = box.x + box.width / 2;
  const y = box.y + box.height / 2;
  await page.mouse.move(x, y);
  await page.mouse.down();
  await page.mouse.move(x, y + dy, { steps: 8 });
  await page.mouse.up();
  await page.waitForTimeout(500);
}
function record(name, checks, extra) {
  const failed = checks.filter(([, ok]) => !ok);
  results.push({ name, checks: checks.map(([n, ok, note]) => ({ n, ok, note })), extra });
  console.log(`\n### ${name}  ${failed.length ? '❌ RED' : '✅ GREEN'}`);
  for (const [n, ok, note] of checks) console.log(`${ok ? 'PASS' : 'FAIL'}  ${n}${note ? `  :: ${note}` : ''}`);
  if (extra) console.log('      extra = ' + JSON.stringify(extra));
  return failed.length === 0;
}
const clamp = (v) => Math.min(1200, Math.max(80, Math.round(v)));

// ── B4：拖拽往返 / 域外 / 防抖持久化 reload / 失败回滚 ─────────────────────────
{
  const page = await openPage('avail=600&page=1');
  const a = await measure(page);
  await drag(page, '15m|1h', 20);
  const b = await measure(page);
  await drag(page, '15m|1h', -20);
  const c = await measure(page);
  await drag(page, '15m|1h', +1000); // 上界域外
  const d = await measure(page);
  await drag(page, '15m|1h', -400); // 下界域外
  const e = await measure(page);
  const pUp = d.payloads[d.payloads.length - 1];
  const pDown = e.payloads[e.payloads.length - 1];
  const inDomain = (p) => ['15m', '1h', '5m', '1d'].every((k) => Number.isInteger(p[k]) && p[k] >= 80 && p[k] <= 1200);
  record('B4a 拖拽往返 + 域外双向夹取（真鼠标拖拽）', [
    ['B4a.1 拖 +20：基准 +20 / 相邻卫星 −20 / Σ 不变', b.domHeights['15m'] === a.domHeights['15m'] + 20 && b.domHeights['1h'] === a.domHeights['1h'] - 20 && b.domSum === 600, `${JSON.stringify(a.domHeights)} → ${JSON.stringify(b.domHeights)}`],
    ['B4a.2 再 −20 精确回到初值（往返）', c.domHeights['15m'] === a.domHeights['15m'] && c.domHeights['1h'] === a.domHeights['1h'] && c.domHeights['5m'] === a.domHeights['5m'], `${JSON.stringify(c.domHeights)} vs ${JSON.stringify(a.domHeights)}`],
    ['B4a.3 上界域外（+1000）⇒ 载荷每项 ∈ [80,1200]', inDomain(pUp), JSON.stringify(pUp)],
    ['B4a.4 下界域外（−400）⇒ 载荷每项 ∈ [80,1200]', inDomain(pDown), JSON.stringify(pDown)],
    ['B4a.5 域外拖拽后 Σ == 600 且 Σ 下界保持', Math.abs(d.domSum - 600) <= 1 && Math.abs(e.domSum - 600) <= 1 && e.domHeights['15m'] >= 200 && Math.min(e.domHeights['1h'], e.domHeights['5m'], e.domHeights['1d']) >= 80, `Σ=${d.domSum}/${e.domSum} ${JSON.stringify(e.domHeights)}`],
    ['B4a.6 每次拖拽恰 1 次写（防抖合并）', d.writes - a.writes === 3 && e.writes - a.writes === 4, `writes=${a.writes}→${d.writes}→${e.writes}`],
    ['B4a.7 main 全程 0 溢出', [a, b, c, d, e].every((x) => x.main.scrollHeight - x.main.clientHeight <= 1)],
  ], { initial: a.domHeights, plus20: b.domHeights, roundTrip: c.domHeights, up: { dom: d.domHeights, payload: pUp }, down: { dom: e.domHeights, payload: pDown }, writes: e.writes });

  // 防抖持久化 → 「重载」保持
  const payload = pUp;
  await page.close();
  const page2 = await openPage(`avail=600&page=1&heights=${encodeURIComponent(JSON.stringify(payload))}`);
  const m2 = await measure(page2);
  record('B4b 防抖持久化 ⇒ 重载后保持（服务端值 = 上次载荷）', [
    ['B4b.1 重载后各 pane 高 == 持久化载荷（±1）', ['15m', '1h', '5m', '1d'].every((k) => Math.abs(m2.domHeights[k] - payload[k]) <= 1), `dom=${JSON.stringify(m2.domHeights)} payload=${JSON.stringify(payload)}`],

  ], { reloadDom: m2.domHeights, payload });
  await page2.close();
}

// ── B5：边界时序（保存成功后再拖）────────────────────────────────────────────
{
  const page = await openPage('avail=1800&page=1');
  const s0 = await measure(page);
  await drag(page, '15m|1h', 1); // 让载荷被夹到 1200（DOM 分配 1261）
  const s1 = await measure(page);
  const p1 = s1.payloads[s1.payloads.length - 1];
  await drag(page, '15m|1h', -20); // 保存成功（回显 1200）后立刻再拖
  const s2 = await measure(page);
  const p2 = s2.payloads[s2.payloads.length - 1];
  await page.close();
  // 「重载」：服务端 = p1（第二次拖拽的载荷与 p1 相同 ⇒ 服务端无变化）
  const page3 = await openPage(`avail=1800&page=1&heights=${encodeURIComponent(JSON.stringify(p1))}`);
  const s3 = await measure(page3);
  await page3.close();
  record('B5 越域下「保存成功后再拖」时序（残余风险实测）', [
    ['B5.1 首次拖 +1：载荷夹到 1200 而 DOM 分配 1261', p1['15m'] === 1200 && s1.domHeights['15m'] > 1200, `payload=1200 dom=${s1.domHeights['15m']}`],
    ['B5.2 二次拖 −20：DOM 变小（视觉有反馈）', s2.domHeights['15m'] < s1.domHeights['15m'], `${s1.domHeights['15m']} → ${s2.domHeights['15m']}`],
    ['B5.3 但载荷与首次完全相同 ⇒ 该拖拽**未被持久化**（死区）', p2['15m'] === p1['15m'] && p2['1h'] === p1['1h'], `p1=${JSON.stringify(p1)} p2=${JSON.stringify(p2)}`],
    ['B5.4 重载后基准 == 1261（≠ 拖后 DOM）⇒ 跳变已实测', s3.domHeights['15m'] === 1261, `reload=${s3.domHeights['15m']} vs 拖后=${s2.domHeights['15m']} ⇒ 跳变 ${s3.domHeights['15m'] - s2.domHeights['15m']}px`],
    ['B5.5 无 pageerror / 无崩溃（行为可预期）', errors.length === 0, errors.join(' | ')],
  ], { s0: s0.domHeights, s1: s1.domHeights, s2: s2.domHeights, s3: s3.domHeights, p1, p2, jumpPx: s3.domHeights['15m'] - s2.domHeights['15m'] });
}
