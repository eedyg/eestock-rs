    pub close_bar: usize,

    /// 成交卖出价（含滑点）。
    pub close_price: f64,
    pub shares: f64,
    /// 卖出毛额（shares × close_price）。
    pub gross_value: f64,

    pub commission: f64,

    pub stamp_duty: f64,

    pub pnl: f64,
    /// 持仓 bar 数（开仓 bar → 平仓 bar 间隔）。
    pub hold_bars: usize,
}
